Hyatt-Mann XOR probe pattern (RESEARCH.md Pattern 1 canonical form):
  bool probe(uint64_t hash, TTEntry& out) {
      AtomicEntry& slot = entries_[hash % num_entries_];
      uint64_t xkey = slot.xkey.load(std::memory_order_relaxed);
      uint64_t data = slot.data.load(std::memory_order_relaxed);
      // XOR validation: any torn read produces (xkey ^ data) != hash → miss (safe)
      if ((xkey ^ data) != hash) {
          miss_count_.fetch_add(1, std::memory_order_relaxed);
          return false;
      }
      unpack_data(data, out.best_move, out.score, out.depth, out.flag, out.age);
      out.key = hash;
      hit_count_.fetch_add(1, std::memory_order_relaxed);
      return true;
  }

Hyatt-Mann XOR store pattern (RESEARCH.md Pattern 1, age-then-depth per D-07):
  void store(uint64_t hash, Move m, int16_t score, int8_t depth, uint8_t flag) {
      AtomicEntry& slot = entries_[hash % num_entries_];
      uint64_t old_xkey = slot.xkey.load(std::memory_order_relaxed);
      uint64_t old_data = slot.data.load(std::memory_order_relaxed);
      uint64_t old_hash = old_xkey ^ old_data;

      bool should_replace;
      if (old_hash == hash) {
          // Same key — always replace (refresh).
          should_replace = true;
      } else {
          uint8_t old_age = uint8_t(old_data >> 48);
          int8_t  old_depth = int8_t(uint8_t(old_data >> 32));
          if (old_age != generation_) {
              // D-07: different age beats same age — always replace stale entries.
              should_replace = true;
          } else {
              // Same age — tie-break on depth.
              should_replace = (depth >= old_depth);
          }
      }
      if (!should_replace) return;

      uint64_t new_data = pack_data(m, score, depth, flag, generation_);
      slot.data.store(new_data, std::memory_order_relaxed);
      slot.xkey.store(hash ^ new_data, std::memory_order_relaxed);
  }
  // NOTE: store order is data-then-xkey so a concurrent probe sees a torn entry as miss.

Stress harness structural skeleton (PATTERNS.md "Stress harness" section — write from scratch):
  // tt_tsan_stress.cpp
  #include "tt.hpp"
  #include <atomic>
  #include <chrono>
  #include <random>
  #include <thread>
  #include <vector>
  int main() {
      v7::TT tt(64);
      std::atomic<bool> stop{false};
      constexpr int N_THREADS = 16;
      constexpr int DURATION_SEC = 60;
      std::vector<std::thread> workers;
      for (int t = 0; t < N_THREADS; ++t) {
          workers.emplace_back([&tt, &stop, t] {
              std::mt19937_64 rng(0xC0FFEEull + t);
              while (!stop.load(std::memory_order_relaxed)) {
                  uint64_t k = rng();
                  if (rng() & 1) {
                      v7::TTEntry e;
                      (void)tt.probe(k, e);
                  } else {
                      tt.store(k, uint16_t(rng() & 0xFFFF),
                               int16_t(rng() & 0xFFFF),
                               int8_t(rng() & 0x7F),
                               uint8_t(v7::TT_EXACT));
                  }
              }
          });
      }
      std::this_thread::sleep_for(std::chrono::seconds(DURATION_SEC));
      stop.store(true, std::memory_order_relaxed);
      for (auto& w : workers) w.join();
      return 0;
  }
