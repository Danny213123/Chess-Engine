namespace v7 {
struct RepStack {
    static constexpr int CAP = 1024;
    uint64_t data[CAP];
    int top = 0;
    void push(uint64_t h) { if (top < CAP) data[top++] = h; }
    void pop()            { if (top > 0) --top; }
    void clear()          { top = 0; }
};
